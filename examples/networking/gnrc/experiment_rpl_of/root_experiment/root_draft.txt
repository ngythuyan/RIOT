#include <string.h>

#include "sema_inv.h"

#include "shell.h"
#include "msg.h"
#define ENABLE_DEBUG 0
#include "debug.h"

#define MAIN_QUEUE_SIZE     (8)
static msg_t _main_msg_queue[MAIN_QUEUE_SIZE];

#define DEFAULT_PORT            (12345)
#define ROOT_ADDRESS            "2001:db8::1"

static sock_udp_t sock;

static char send_thread_stack[THREAD_STACKSIZE_DEFAULT];
static char listen_thread_stack[THREAD_STACKSIZE_DEFAULT];

static uint32_t delay_us = US_PER_SEC;
static bool running;
static sema_inv_t thread_sync;

static void _print_addr(void) 
{
    /* print all IPv6 addresses */
    printf("{\"IPv6 addresses\": [\"");
    netifs_print_ipv6("\", \"");
    puts("\"]}");
}

static void *_listen_thread(void *ctx)
{
    static char server_buffer[128];
    
    while (1) {
        /* receive UDP */
        int res;

        if ((res = sock_udp_recv(&sock, server_buffer,
                                sizeof(server_buffer) - 1, SOCK_NO_TIMEOUT,
                                NULL)) < 0) {
            printf("Error while receiving: %d\n", res);
        }
        else if (res == 0) {
            puts("No data received");
        }
        else {
            printf("Received: %.*s\n", res, server_buffer);
        }
    }

    (void)ctx;

    DEBUG_PUTS("listen thread start");

    while (running) {
        // receive msg

        unsigned state = irq_disable();
        // verarbeiten
        irq_restore(state);
    }

    DEBUG_PUTS("bench_udp: listen thread terminates");
    sema_inv_post(&thread_sync);

    return NULL;
}

static void *_send_thread(void *ctx)
{
    sock_udp_ep_t remote = *(sock_udp_ep_t*)ctx;

    char buf[IPV6_ADDR_MAX_STR_LEN];
    ipv6_addr_to_str(buf, (ipv6_addr_t *)remote.addr.ipv6, sizeof(buf));
    printf("Sending thread: sending to [%s]:%u on netif %u\n", buf, remote.port, remote.netif);

    DEBUG_PUTS("sending thread start");

    while (running) {
        char message[] = "Ping from root\0";

        /* prepare udp endpoint*/
        int res;

        /* send UDP msg */
        if((res = sock_udp_send(NULL, message, sizeof(message), &remote)) < 0) {
            puts("could not send");
        }

        if (sock_udp_send(&sock, message, sizeof(message), &remote) < 0) {
            puts("Error sending message");
            continue;
        } else {
            unsigned state = irq_disable();
            // do stuff
            irq_restore(state);
        }

        xtimer_usleep(delay_us);
    }

    DEBUG_PUTS("bench_udp: sending thread terminates");
    sema_inv_post(&thread_sync);

    return NULL;
}

/* ================= Main ================= */
int main(void)
{
    printf("\n=== RPL Root Starting ===\n");
    xtimer_sleep(2);

    /* ================= Init Root ================= */
    /* Get first network interface */
    gnrc_netif_t *netiface = gnrc_netif_iter(NULL);
    if (!netiface) {
        puts("No network interface found");
        return 1;
    }
    printf("Network interface PID: %d\n", netiface->pid);

    /* init RPL */
    gnrc_rpl_init(netiface->pid);
    xtimer_sleep(2);

    /* add IPv6 address */
    ipv6_addr_t dodag_id;
    ipv6_addr_from_str(&dodag_id, ROOT_ADDRESS);
    gnrc_netif_ipv6_addr_add(netiface, &dodag_id, 64, 0);

    _print_addr();

    /* init RPL root */
    gnrc_rpl_instance_t *inst = gnrc_rpl_root_init(1, &dodag_id, false, false);
    if (!inst) {
        puts("RPL Root init failed");
        return 1;
    }

    /* UDP setup */
    netif_t *netif;
    sock_udp_ep_t local = { .family = AF_INET6,
                            .netif = SOCK_ADDR_ANY_NETIF,
                            .port = DEFAULT_PORT };
    sock_udp_ep_t remote = { 0 };
    remote.family = AF_INET6;
    remote.port = DEFAULT_PORT;

    if (sock_udp_create(&sock, &local, NULL, 0) < 0) {
        puts("Error creating UDP sock");
        return 1;
    }

    /* ================= receive messages ================= */
    running = true;
   thread_create(listen_thread_stack, sizeof(listen_thread_stack),
                 THREAD_PRIORITY_MAIN - 2, 0,
                 _listen_thread, NULL, "UDP receiver");

   puts("Created listening thread");
    /* ================= send messages ================= */
   thread_create(send_thread_stack, sizeof(send_thread_stack),
                 THREAD_PRIORITY_MAIN - 1, 0,
                 _send_thread, &remote, "UDP sender");

   puts("Created sending thread");

    /* ================= start shell ================= */
    puts("Running Shell");
    msg_init_queue(_main_msg_queue, MAIN_QUEUE_SIZE);
    char line_buf[SHELL_DEFAULT_BUFSIZE];
    shell_run(NULL, line_buf, SHELL_DEFAULT_BUFSIZE);

    return 0;
}
